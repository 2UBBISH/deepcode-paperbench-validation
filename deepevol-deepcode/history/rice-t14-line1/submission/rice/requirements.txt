# ------------------- environment stack (required for MuJoCo) -----------------
# Legacy gym API (Hopper-v3 / Walker2d-v3 / Reacher-v2 / HalfCheetah-v3 ids).

# re-injects the healthy reward removed in Gymnasium's v4 rewrite.

# ------------------------- optional / in-scope extras ------------------------
# PPO/SAC reference implementations (the code ships self-contained PPO and SAC,
# but SB3 is used when available to obtain the exact SB3 default 'MlpPolicy').
stable-baselines3>=1.7      # SB3 defaults for PPO (lr 3e-4, n_steps 2048, ...)
