#     pip install -r requirements.txt

# further down and can be installed incrementally.

# ---------------------------------------------------------------------------
# Deep learning
# ---------------------------------------------------------------------------
# PyTorch (Paszke et al. 2019) -- mask network (Stage 1) and PPO refinement

#   CUDA: see https://pytorch.org/get-started/previous-versions/
torch>=1.13,<2.4
